            let nextM = gameState.calendar.find(m => !m.played);
            if(nextM) {
                document.getElementById('next-match-info').innerText = `${nextM.comp}: ${gameState.teamName} vs ${nextM.opponent} (Semaine ${nextM.week})`;
            } else {
                document.getElementById('next-match-info').innerText = "Saison Terminée — Cérémonie du Ballon d'Or !";
            }
        }

        function renderSquad() {
            let html = "";
            gameState.squad.forEach(p => {
                html += `<tr>
                    <td><strong>${p.name}</strong></td>
                    <td>${p.pos}</td>
                    <td>${p.gen}</td>
                    <td>${p.form}%</td>
                    <td>${p.moral}</td>
                    <td>${p.val}</td>
                    <td><button class="btn" style="padding: 4px 10px; font-size: 0.8rem;" onclick="talkToPlayer('${p.name}')">Discuter</button></td>
                </tr>`;
            });
            document.getElementById('squad-table-body').innerHTML = html;
        }

        function talkToPlayer(playerName) {
            alert(`Vous avez une discussion constructive avec ${playerName}. Son moral s'améliore et son implication tactique augmente !`);
            gameState.teamForm = Math.min(100, gameState.teamForm + 2);
            updateHUD();
            renderSquad();
        }

        function updateTacticalSelection() {
            let sel = document.getElementById('active-tactic').value;
            gameState.activeTactic = sel;
            alert(`Tactique définie : ${sel}. L'équipe va l'assimiler à l'entraînement.`);
        }

        function runTrainingSession() {
            let schedType = document.getElementById('training-schedule-type').value;
            // Manuel V4.1.2 application
            gameState.fitness = Math.min(100, gameState.fitness + 4);
            gameState.tacticalMastery = Math.min(100, gameState.tacticalMastery + (schedType === "1" ? 15 : 10));
            gameState.teamForm = Math.min(100, gameState.teamForm + 5);

            let logBox = document.getElementById('training-result-box');
            logBox.innerHTML = `✅ Semaine d'entraînement V4.1.2 validée avec succès !<br>• Application des mots-clés (LARGEUR, SUPÉRIORITÉ, 3 SECONDES).<br>• Maîtrise tactique actuelle : ${gameState.tacticalMastery}%<br>• Condition physique optimisée à ${gameState.fitness}%.`;
            updateHUD();
        }

        function renderCalendar() {
            let html = "";
            gameState.calendar.forEach(m => {
                let statusBadge = m.played ? '<span style="color: #10b981;">Joué</span>' : '<span style="color: var(--secondary);">À jouer</span>';
                html += `<tr><td>Semaine ${m.week}</td><td>${m.comp}</td><td>${m.opponent}</td><td>${statusBadge}</td></tr>`;
            });
            document.getElementById('calendar-table-body').innerHTML = html;
        }

        function simulateWeek() {
            gameState.week++;
            if(gameState.week > 38) {
                alert("Fin de la saison ! Cérémonie du Ballon d'Or et distribution des trophées.");
                gameState.trophies.push({ name: "Championnat National / Trophée S4", season: gameState.season });
                renderTrophies();
                gameState.week = 1;
                gameState.season++;
            }
            updateHUD();
        }

        function answerPress(choice) {
            if(choice === 1) {
                alert("Conférence réussie : L'équipe est pleinement concentrée et confiante.");
                gameState.teamForm += 3;
            } else {
                alert("Conférence correcte : Pression mise sur les individualités.");
                gameState.reputation += 1;
            }
            updateHUD();
        }

        function makeOffer(playerName, price) {
            if(gameState.budget >= price) {
                gameState.budget -= price;
                alert(`Offre acceptée ! ${playerName} rejoint votre effectif après une négociation de contrat réussie.`);
                updateHUD();
            } else {
                alert("Budget insuffisant pour recruter ce joueur.");
            }
        }

        function renderTrophies() {
            let container = document.getElementById('trophy-room-list');
            let html = "";
            gameState.trophies.forEach(t => {
                html += `<div style="background: rgba(0,240,255,0.1); border: 1px solid var(--secondary); padding: 15px 25px; border-radius: 8px; text-align: center;">
                    <div style="font-size: 2rem;">🏆</div>
                    <div style="font-weight: bold; margin-top: 5px;">${t.name}</div>
                    <div style="font-size: 0.8rem; color: #8a99ad;">Saison ${t.season}</div>
                </div>`;
            });
            container.innerHTML = html;
        }

        // MATCH ENGINE WITH DUAL COMMENTATORS & MUSIC SWITCH
        function launchNextMatch() {
            let nextM = gameState.calendar.find(m => !m.played);
            if(!nextM) {
                alert("Plus de matchs prévus pour cette saison.");
                return;
            }

            // Arrêt de la musique de fond et lancement du match
            stopBackgroundMusic();

            document.getElementById('main-game').style.display = 'none';
            document.getElementById('match-engine').style.display = 'flex';
            document.getElementById('match-team-home').innerText = gameState.teamName;
            document.getElementById('match-team-away').innerText = nextM.opponent;

            let homeScore = 0;
            let awayScore = 0;
            let minute = 0;
            let log = document.getElementById('commentary-log');
            log.innerHTML = `[0'] Coup d'envoi de ${nextM.comp} entre ${gameState.teamName} et ${nextM.opponent} !<br>`;

            // Double commentateurs (Homme & Femme alternés)
            let maleComments = [
                "Quelle intensité au milieu de terrain ! Nos joueurs appliquent parfaitement la tactique.",
                "Superbe interception, la transition offensive est instantanée.",
                "Frappe lointaine captée sans problème par le gardien adverse."
            ];
            let femaleComments = [
                "L'intelligence de placement des joueurs fait toute la différence aujourd'hui !",
                "Quelle magnifique inspiration technique dans le demi-espace !",
                "Le bloc défensif est bien compact, l'adversaire étouffe."
            ];

            let matchInterval = setInterval(() => {
                minute += 5;
                if(minute > 90) {
                    clearInterval(matchInterval);
                    log.innerHTML += `<br><strong>[90'] C'est terminé ! Score final : ${homeScore} - ${awayScore}</strong><br>`;
                    nextM.played = true;
                    setTimeout(() => {
                        document.getElementById('match-engine').style.display = 'none';
                        document.getElementById('main-game').style.display = 'flex';
                        playBackgroundMusic(); // Reprise de la musique
                        updateHUD();
                        renderCalendar();
                    }, 4000);
                    return;
                }

                // Animation du ballon sur la pelouse
                let ball = document.getElementById('match-ball');
                ball.style.top = (20 + Math.random() * 60) + "%";
                ball.style.left = (10 + Math.random() * 80) + "%";

                if(Math.random() < 0.35) {
                    if(Math.random() > 0.5) {
                        homeScore++;
                        log.innerHTML += `[${minute}'] ⚽ <strong>BUUUT pour ${gameState.teamName} !</strong> (${maleComments[Math.floor(Math.random()*maleComments.length)]})<br>`;
                    } else {
                        awayScore++;
                        log.innerHTML += `[${minute}'] ⚽ But pour ${nextM.opponent}. (${femaleComments[Math.floor(Math.random()*femaleComments.length)]})<br>`;
                    }
                    document.getElementById('match-score-display').innerText = `${homeScore} - ${awayScore}`;
                }
                log.scrollTop = log.scrollHeight;
            }, 800);
        }
    </script>
